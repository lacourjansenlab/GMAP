36 def __eq__(self, other): 

37 """This is used to see if two error codes equal each other. 

38 

39 This function exists to quickly check if an error code is 

40 contained within a container (that should loop and _eq_ each 

41 item, if I understand correctly). 

42 

43 One of the error codes can be a string (but not both, because 

44 in that case this method would not be called). 

45 

46 Examples 

47 -------- 

48 >>> errcode = ErrCode("AA_BB_33") 

49 

50 >>> errcode == "AA_BB_33" 

51 True 

52 >>> errcode == "AA_BC_33" 

53 False 

54 

55 If one of the two in the comparison is a wildcard (nothing 

56 specified in that part), that part will always equal. Note the 

57 double underscore for a wildcard in the middle field: 

58 

59 >>> errcode == "AA_BB_" 

60 True 

61 >>> errcode == "AA_BC_" 

62 False 

63 >>> errcode == "AA__33" 

64 True 

65 """ 

66 

67 # type checking 

68 selfsplit = self.split("_") 

69 if len(selfsplit) != 3: 

70 GM_pt.Printer.warning( 

71 f"\n{self} is assumed to be an error code, but does not have " 

72 "3 parts separated by underscores. Please make sure to only " 

73 "compare valid error codes.", 

74 "CT_EC_2", True, GMAPerrclass=GM_ex.GmapValueError 

75 ) 

76 

77 if not isinstance(other, str): 

78 GM_pt.Printer.warning( 

79 f"\n{other} is assumed to be an error code, but is not a " 

80 "string, so this method cannot be used. Please make sure to " 

81 "only compare strings or ErrCodes.", 

82 "CT_EC_1", True, GMAPerrclass=GM_ex.GmapTypeError 

83 ) 

84 othersplit = other.split("_") 

85 if len(othersplit) != 3: 

86 GM_pt.Printer.warning( 

87 f"\n{other} is assumed to be an error code, but does not have " 

88 "3 parts separated by underscores. Please make sure to only " 

89 "compare valid error codes.", 

90 "CT_EC_2", True, GMAPerrclass=GM_ex.GmapValueError 

91 ) 

92 

93 # check the actual equality 

94 matched = 0 

95 for selfsub, othersub in zip(selfsplit, othersplit): 

96 if "" in (selfsub, othersub): 

97 matched += 1 

98 elif selfsub == othersub: 

99 matched += 1 

100 

101 if matched == 3: 

102 return True 

103 else: 

104 return False

107class ColStr(str): 

108 """Just like normal strings, but additional methods for colors. 

109 

110 If any string is added to this class (either from the left or from 

111 the right), the result will be of this class. Similarly, many other 

112 str operations are possible with this class. When properly 

113 supported, they will return a ColStr, not 'normal' str. 

114 

115 So, what's the difference? Consider the following string: 

116 ``Hello \033[35;1m pretty \033[0m world`` 

117 It contains two ANSI sequences. In this case, they indicate that the 

118 word 'pretty' should be printed in bright magenta, whereas 'Hello' 

119 and 'world' should be printed in 'default' colors. As a normal 

120 python string, it contains 31 characters: all the letters, and all 

121 the characters of the ANSI codes (the 033 thing is a single 

122 character). So, its length is 31, and the for-loop iterating through 

123 it will have 31 iterations. However, when printed to the command 

124 line, it will only take up 20 characters. 

125 

126 When dealing with prints, it is very nice to deal with the length 

127 of the string as observed in the command line. All the functions 

128 of this class work that way, or are designed to return an instance 

129 of ColStr, instead of str. 

130 

131 Currently supported methods: 

132 __add__, __radd__, __getitem__ (slicing/indexing), __iter__ (for 

133 item in, list(ColStr)), __len__, __mul__, __rmul__, join, 

134 replace, split 

135 

136 .. warning:: 

137 str.join([ColStr]) will result in a str object, not ColStr! 

138 

139 .. warning:: 

140 The color markers are 'attached' to the following character (if 

141 there is any). If that character is sliced out, the color will 

142 disappear. If this behaviour poses an issue, open an issue for 

143 it on github! 

144 """ 

145

152 def __getitem__(self, key): 

153 if isinstance(key, int): # indexing 

154 if key >= len(self): 

155 raise GM_ex.GmapIndexError 

156 key = (key, key + 1, 1) 

157 return ColStr("").join([*self._forwards_generator(key)]) 

158 

159 elif isinstance(key, slice): # slicing 

160 key = key.indices(len(self)) # makes sure start and stop >= 0 

161 if key[2] > 0: 

162 return ColStr("").join([*self._forwards_generator(key)]) 

163 else: 

164 # Haven't implemented backwards yet, as it might make less 

165 # sense for colored strings. Multiple options: 

166 # - Same as forward, match color with character -> colors get 

167 # funky 

168 # - For each character, identify intended color, and make sure 

169 # those are preserved -> actual sequence changes 

170 # - Omit colors and just revert the 'white' string. 

171 

172 # key = (key[1], key[0], -1 * key[2]) 

173 raise GM_ex.GmapNotImplementedError

187 def _forwards_generator(self, key): 

188 """A generator in the forwards direction. 

189 

190 This is the backbone of __iter__ and __getitem__ -> this is the 

191 actual mechanism of stepping through the string. Due to the 

192 logic of recognizing color markers, this exact function only 

193 works forwards, not backwards. See comments of the __getitem__ 

194 method for more info on backwards. 

195 

196 Parameters 

197 ---------- 

198 key : tuple 

199 This tuple contains three integers: start, stop (exclusive) 

200 and step. Due to the creation of key, start and stop are 

201 always positive integers. If a negative-based slice was 

202 requested, the form was already changed to positive integers 

203 before being fed here. 

204 

205 Returns 

206 ------- 

207 char : `ColStr` 

208 The next character in the sequence. When printed to the 

209 terminal, it will appear to have a length of 1 (could also 

210 be a newline character), but len(str(char)) might not be 1: 

211 it could contain an ANSI color marker! 

212 """ 

213 

214 col_ix = 0 

215 within_col = False 

216 col_end = "" 

217 tempstr = "" 

218 

219 for item in str(self): # loop over all characters 

220 # Stop if we went past endpoint 

221 if col_ix >= key[1]: 

222 break 

223 

224 # within color marker? skip through! 

225 if within_col: # add item to bin, check if still going 

226 if len(tempstr) == 1: 

227 col_end = "m" if item == "[" else ">" 

228 tempstr += item 

229 if item == col_end: 

230 within_col = False 

231 continue 

232 

233 # start of color marker! 

234 if item == "\033": # found a new color 

235 within_col = True 

236 tempstr += item 

237 continue 

238 

239 # now, a non-color character! 

240 if len(tempstr) != 0: # if prev was a color, add it! 

241 item = tempstr + item 

242 col_end = "" 

243 tempstr = "" 

244 

245 # must it be reported? 

246 if col_ix < key[0]: # if its too small/soon 

247 pass 

248 elif col_ix == key[0]: 

249 yield item 

250 # key[0] < item < key[1], and at step. 

251 elif (col_ix - key[0]) % key[2] == 0: 

252 yield item 

253 col_ix += 1

307 def change_color(self, target_mode): 

308 """Changes its own color, returns a copy with changed color 

309 

310 There are different types of colors: internal colors, ANSI 

311 24bit, ANSI 4bit, and omitting all color. When this function is 

312 called, any internals present are first converted to 24bit. 

313 Then, all colors are converted to target_mode. 

314 

315 Internal colors: We want to be able to give all prints from the 

316 program a color, even the first. However, at that point in the 

317 program, we don't yet know whether to print in dark, or light 

318 mode. Internal colors work with color names which are converted 

319 to actual ANSI strings when this function runs. 

320 

321 24bit colors: each color has a rgb code. This is the default of 

322 the program (all colors are defined that way), and allows for 

323 the prettiest output. 

324 

325 4bit colors: These are so old, any hardware imaginable should 

326 support them. Their larger contrast (as there are so few) should 

327 also help vision-impaired folks. 

328 

329 no color: all color markers are removed. Quite useful for when 

330 writing to the log file. 

331 

332 Parameters 

333 ---------- 

334 target_mode : str 

335 Either '24bit', '4bit' or 'white'. The kind of colors you 

336 want to see. 

337 

338 Returns 

339 ------- 

340 output : str 

341 The same string as the input one, but with the color markers 

342 changed to the desired output, or removed altogether. 

343 """ 

344 

345 # The code should be able to choose the color scheme last-minute, so 

346 # there are internal codes, too! Here, we switch from internal to ANSI 

347 # Even if there are no custom markers, don't quit, there might be ANSI 

348 # markers still! 

349 colstr = self 

350 colstr_split = self.split("\033<") 

351 if len(colstr_split) != 1: # custom color marker! 

352 string_list = [item.split(">", 1) for item in colstr_split] 

353 newlist = [">".join(string_list[0])] # the '>' isn't color-closing 

354 for item in string_list[1:]: 

355 newlist.append(getattr( 

356 GM_pt.Printer._colors, item[0]) + item[1]) 

357 colstr = ColStr("").join(newlist) 

358 

359 # If we want 24bit, we can stay with the current ANSI codes. 

360 if target_mode == "24bit": 

361 return colstr 

362 

363 # make sure the beginning text has a color, too (just for code 

364 # simplification, not actually in output) 

365 colstr_split = colstr.split("\033[") 

366 if len(colstr_split) == 1: # no color marker! 

367 return colstr 

368 

369 # prepare the string - split it up into a list in which each item 

370 # represents a monocolor segment. The item is a list of length 2, first 

371 # the ansi color string, then the actual string. 

372 newstr = "0m" + colstr 

373 colstr_split = newstr.split("\033[") 

374 # cut only at first m as thats the end of the color marker 

375 colstr_list = [item.split("m", 1) for item in colstr_split] 

376 

377 # now, change the color of each monocolor substring 

378 if target_mode == "4bit": # 8 colors + their bright varieties 

379 output = colstr_list[0][1] # don't keep color of first item 

380 for item in colstr_list[1:]: 

381 col = self.ANSI24_to_ANSI4(item[0]) 

382 output += f"\033[{col}m{item[1]}" 

383 elif target_mode == "white_str": # we need output just for len 

384 output = "".join([item[1] for item in colstr_list]) 

385 else: # target_mode white 

386 # just delete any color markers 

387 output = ColStr("").join([item[1] for item in colstr_list]) 

388 return output

390 def ANSI24_to_ANSI4(self, colorstr): # used by change_color 

391 """Change a ANSI-24bit color code to the closest 4bit version. 

392 

393 Input colorstr in ANSI format (e.g. 38;2;45;61;32). Both 

394 background and foreground colors are treated. For backgrounds, 

395 the 'bright' bit is never activated (not all terminals support 

396 it), but it is for foreground. 

397 """ 

398 

399 warning_msg = "\nInvalid color specification." 

400 

401 color_split = colorstr.split(";") # get all useful values 

402 color_new = [] 

403 while len(color_split) > 0: 

404 match color_split[0]: 

405 case "0": # reset command - no extra's expected 

406 color_new.append("0") 

407 color_split = color_split[1:] 

408 

409 case "38": # foreground - 5 items including this one 

410 curr_color = color_split[:5] 

411 if len(curr_color) != 5: # premature end of list 

412 GM_pt.Printer.warning(warning_msg, "PT_CC_1", True) 

413 

414 new_col, bright = GM_mf.convert_color_24_4(*curr_color[2:]) 

415 color_new.append(str(30 + new_col)) 

416 if bright: 

417 color_new.append("1") 

418 color_split = color_split[5:] 

419 

420 case "48": # background - 5 items including this one 

421 curr_color = color_split[:5] 

422 if len(curr_color) != 5: # premature end of list 

423 GM_pt.Printer.warning(warning_msg, "PT_CC_1", True) 

424 

425 new_col = GM_mf.convert_color_24_4(*curr_color[2:])[0] 

426 color_new.append(str(40 + new_col)) 

427 # no bright - a bright background is not possible 

428 color_split = color_split[5:] 

429 

430 case _: 

431 GM_pt.Printer.warning(warning_msg, "PT_CC_1", True) 

432 

433 return ";".join(color_new)

435 def repeat_color(self): 

436 """Makes sure to repeat earlier color codes after line breaks. 

437 

438 Input strings MUST contain ANSI sequences, any others are 

439 ignored. 

440 """ 

441 

442 # If there is no line break, no need to repeat colors 

443 if "\n" not in self: 

444 return self 

445 

446 # Split the string on color markers first. 

447 colstr_split = self.split("\033[") 

448 if len(colstr_split) == 1: # no color marker 

449 return self 

450 

451 # the first part can be directly outputted. Because of the 

452 # implementation of split, it is already of type ColStr. It maintains 

453 # its type through additions 

454 outcolstr = colstr_split[0] 

455 # Check between any two color markers (and after the last) 

456 for substr in colstr_split[1:]: 

457 color, text = substr.split("m", 1) # separate color and message 

458 # Anytime a newline occurs, repeat the color after it. Also add the 

459 # color to the beginning of this string (where it was stripped away 

460 # from) 

461 color = f"\033[{color}m" 

462 outcolstr += color + text.replace("\n", f"\n{color}") 

463 

464 return outcolstr

467class Header: 

468 """Returns a pretty header for distinguishing prints 

469 

470 The header will be under and/or overlined with the character 

471 buffer_char. These lines will have the same length as the title, 

472 plus extra padding if requested. 

473 

474 Parameters 

475 ---------- 

476 title : str 

477 The text that should be within the header. 

478 padding_char : str 

479 The character that should be used for padding around the title 

480 linemode : str, default="ou" 

481 which lines should(n't) be displayed. Add the letters for the 

482 elements you do want to display:: 

483 

484 o: A line above the text 

485 u: A line below the text 

486 l: A line left of the text 

487 r: A line right of the text 

488 c: Corners 

489 

490 padding : int, default=0 

491 How much extra space there should be. Over and underlines will 

492 get longer by twice this amount, the title itself gets this 

493 amount of whitespaces before and after the text. 

494 overline_char, underline_car, left_char, right_char : str, \ 

495 default="-" 

496 The character that should be used for the respective segment 

497 corner_char : str, default="+" 

498 The character that should be used for the corners. If a single 

499 character is provided, all corners get that character. If 

500 different characters are desired, there must be 4 characters 

501 in the string: one for the upleft, upright, downleft, downright 

502 corner respectively. 

503 alignment : str, default="centered" 

504 If the title text consists of multiple lines of unequal length, 

505 decide how these should be aligned. Regardless of choice, the 

506 character specified using padding_char will be used to increase 

507 the lengths of shorter lines. The amount of characters specified 

508 using padding will be added to the lines after they are made of 

509 equal length. 

510 "centered" aligns their midpoints, "leftadj" will align their 

511 left edges, and "rightadj" will align their right ones. 

512 maxwidth : int, default=79 

513 How wide the end result is allowed to be, max. If the title 

514 text is too wide to fit the requirements, it is line-wrapped to 

515 fit. 

516 special : dict, default={} 

517 Any special rules to apply after the header has been generated. 

518 Thakes the form of a nested dictionary structure: 

519 special={"u": {"command": {"|": [[2]]}}} 

520 

521 - The outermost dict has as the key a string indicating what 

522 line should be altered. Same shorthands as linemode. The value 

523 of the dictionary is what should be changed about the line 

524 (here referred to as the 'rules') 

525 - The rules dict has the commands as keys, and specific 

526 instructions (another dict) as the values: 

527 

528 - replace will put the character saved as key in the 

529 instructions dict at the positions saved in its values: 

530 if the line is "abcdefg", {"replace": ["V", [[3]]]} will 

531 result in "abcVefg". Similarly, {"replace": ["VR", 

532 [[1, 3], [4, 6]]]} will turn "abcdefg" into "aVRdVRg" 

533 

534 Returns 

535 ------- 

536 header : str 

537 The header, ready for printing. 

538 

539 Attributes 

540 ---------- 

541 title : `ColStr` or list of `ColStr` 

542 The text to be displayed in the header. At the start of __init__ 

543 this is a ColStr, and gets converted to a list of ColStr during 

544 execution of __init__. 

545 header : `ColStr` 

546 The complete header, including all requested features. This is 

547 the actual string that the caller most likely will want to 

548 print. 

549 s : str 

550 The output of str(self) 

551 title_lines : int 

552 The amount of lines the tile spans 

553 longest_length : int 

554 The amount of characters the longest line of the title spans. 

555 over : `ColStr` 

556 The overline to put in the header (also contains two upper 

557 corners if corners have been requested) 

558 under : `ColStr` 

559 The underline to put in the header (also contains two lower 

560 corners if corners have been requested) 

561 left : list of `ColStr` 

562 The left border line to put in the header 

563 right : list of `ColStr` 

564 The right border line to put in the header 

565 """

612 def format_title(self, maxwidth, n_padding): 

613 """Prepare the title text for the header 

614 

615 Provided with the maximum width of the entire header, figures 

616 out the maximum line length for the text within, and adjusts 

617 self.title to match. Also saves how many lines it created/found. 

618 

619 Parameters 

620 ---------- 

621 maxwidth : int 

622 How wide the end result is allowed to be, max. If the title 

623 text is too wide to fit the requirements, it is line-wrapped 

624 to fit. 

625 n_padding : int 

626 How much extra space there should be. Over and underlines 

627 will get longer by twice this amount, the title itself gets 

628 this amount of whitespaces before and after the text. 

629 """ 

630 

631 # line wrapping at the correct places. 

632 title_width = maxwidth - 2 - (n_padding * 2) # leave space for header 

633 self.title = self.title.wrap(title_width) 

634 self.title_lines = len(self.title.split("\n"))

636 def align_title(self, alignment, padding_char): 

637 """Adds whitespaces to all title lines to make them equal length 

638 

639 When adding these whitespaces, for each line, it must be decided 

640 whether to add them to the beginning, end, or both. This 

641 function can either centre all lines, or left- or right adjust 

642 them. 

643 

644 Parameters 

645 ---------- 

646 alignment : str 

647 How to vertically align the lines. Can be 'leftadj', 

648 'rightadj' or 'centered'. 

649 padding_char : str or `ColStr` 

650 The character used to make the shortest lines longer. 

651 """ 

652 

653 # padding and adjusting the lines 

654 title = self.title.split("\n") 

655 self.longest_length = max([len(item) for item in title]) 

656 if self.title_lines == 1: 

657 self.title = title 

658 return 

659 

660 # doing it like this instead of python str.centre, rjust, ljust, as 

661 # those don't take the colors into account correctly! 

662 new_title = [] 

663 for line in title: 

664 diff = self.longest_length - len(line) 

665 match alignment: 

666 case "centered": 

667 pre = diff // 2 

668 post = diff - pre 

669 case "leftadj": 

670 pre = 0 

671 post = diff 

672 case "rightadj": 

673 pre = diff 

674 post = 0 

675 new_title.append(pre * padding_char + line + post * padding_char) 

676 self.title = new_title 

677 return

679 def make_lines( 

680 self, linemode, n_padding, overline_char="-", underline_char="-", 

681 left_char="|", right_char="|", corner_char="+" 

682 ): 

683 """Creates the lines making up the border. 

684 

685 Parameters 

686 ---------- 

687 linemode : str, default="ou" 

688 which lines should(n't) be displayed. Add the letters for 

689 the elements you do want to display:: 

690 

691 o: A line above the text 

692 u: A line below the text 

693 l: A line left of the text 

694 r: A line right of the text 

695 c: Corners 

696 n_padding : int, default=0 

697 How much extra space there should be. Over and underlines 

698 will get longer by twice this amount, the title itself gets 

699 this amount of whitespaces before and after the text. 

700 overline_char, underline_car, left_char, right_char : str, \ 

701 default="-" 

702 The character that should be used for the respective segment 

703 corner_char : str, default="+" 

704 The character that should be used for the corners. If a 

705 single character is provided, all corners get that 

706 character. If different characters are desired, there must 

707 be 4 characters in the string: one for the upleft, upright, 

708 downleft, downright corner respectively. 

709 """ 

710 

711 ou_length = self.longest_length + n_padding * 2 

712 corner_char = corner_char * 4 # automatically solves all! 

713 

714 # overline 

715 self.over = self.line_overunder( 

716 overline_char, corner_char[:2], linemode, ou_length, "o") 

717 

718 # sidelines 

719 self.left = self.line_sides(left_char, linemode, self.title_lines, "l") 

720 self.right = self.line_sides( 

721 left_char, linemode, self.title_lines, "r") 

722 

723 # underline 

724 self.under = self.line_overunder( 

725 underline_char, corner_char[2:4], linemode, ou_length, "u")

727 def line_overunder(self, line_char, corner_chars, linemode, length, dir): 

728 """The logic for creating over and underlines. 

729 

730 Parameters 

731 ---------- 

732 line_char : str 

733 The character that should be used to create the line 

734 corner_chars : str of length 2 

735 The two corners that should go at the beginning and end of 

736 the line, if requested. 

737 linemode : str 

738 which lines should(n't) be displayed. Add the letters for 

739 the elements you do want to display:: 

740 

741 o: A line above the text 

742 u: A line below the text 

743 l: A line left of the text 

744 r: A line right of the text 

745 c: Corners 

746 length : int 

747 How long the line-part of the segment should be. 

748 dir : str 

749 What direction the line is being made for. Can be 'o' or 

750 'u', using the same codes as linemode. 

751 

752 Returns 

753 ------- 

754 line : `ColStr` 

755 The line as should be put above/below the header. 

756 """ 

757 

758 # Get middle portion (if no overline, but corners, we need whitespace) 

759 if dir in linemode: 

760 line = line_char * length 

761 elif "c" in linemode: 

762 line = " " * length 

763 else: 

764 line = "" 

765 

766 # add corners! 

767 if "c" in linemode: 

768 line = corner_chars[0] + line + corner_chars[1] 

769 elif dir not in linemode: 

770 pass # if the entire line is empty, no need to add vertical spacer 

771 elif "l" in linemode or "r" in linemode: 

772 if "l" in linemode: 

773 line = " " + line 

774 if "r" in linemode: 

775 line += " " 

776 

777 return ColStr(line)

779 def line_sides(self, line_char, linemode, length, dir): 

780 """The logic for creating left and right lines. 

781 

782 Parameters 

783 ---------- 

784 line_char : str 

785 The character that should be used to create the line 

786 linemode : str 

787 which lines should(n't) be displayed. Add the letters for 

788 the elements you do want to display:: 

789 

790 o: A line above the text 

791 u: A line below the text 

792 l: A line left of the text 

793 r: A line right of the text 

794 c: Corners 

795 length : int 

796 How long the line-part of the segment should be. 

797 dir : str 

798 What direction the line is being made for. Can be 'l' or 

799 'r', using the same codes as linemode. 

800 

801 Returns 

802 ------- 

803 line : list of `ColStr` or list of str 

804 The line as should be put above/below the header. Each item 

805 in the list is what should go on a separate line. Lists 

806 are needed in the case a line (and the corners) are not 

807 requested - then we need to have an empty string for that 

808 line. 

809 """ 

810 

811 if dir in linemode: 

812 line = [line_char[0]] * length 

813 elif "c" in linemode: 

814 line = [" "] * length 

815 else: 

816 line = [""] * length 

817 return line

819 def do_specials(self, special): 

820 """Apply the requested specials to the created lines. 

821 

822 This allows to make small edits to the given lines. 

823 

824 Parameters 

825 ---------- 

826 special : dict, default={} 

827 Any special rules to apply after the header has been 

828 generated. Thakes the form of a nested dictionary structure: 

829 special={"u": {"command": {"|": [[2]]}}} 

830 

831 - The outermost dict has as the key a string indicating what 

832 line should be altered. Same shorthands as linemode. The 

833 value of the dictionary is what should be changed about 

834 the line (here referred to as the 'rules') 

835 - The rules dict has the commands as keys, and specific 

836 instructions (another dict) as the values: 

837 

838 - replace will put the character saved as key in the 

839 instructions dict at the positions saved in its values: 

840 if the line is "abcdefg", {"replace": ["V", [[3]]]} will 

841 result in "abcVefg". Similarly, {"replace": ["VR", 

842 [[1, 3], [4, 6]]]} will turn "abcdefg" into "aVRdVRg" 

843 """ 

844 

845 shorts = {"o": "over", "u": "under", "l": "left", "r": "right"} 

846 

847 for lineshort, rules in special.items(): 

848 line = getattr(self, shorts[lineshort]) 

849 for command, instructions in rules.items(): 

850 if command == "replace": 

851 line = self.specials_replace( 

852 lineshort, instructions, line) 

853 setattr(self, shorts[lineshort], line) 

854 

855 def specials_replace(self, lineshort, instructions, line): 

856 """Executes the specials option 'replace' 

857 

858 Parameters 

859 ---------- 

860 lineshort : str 

861 The shorthand representing what line is currently fed to 

862 the function - 'o', 'u', 'l', or 'r' 

863 instructions : dict 

864 The dict saved in specials for the key 'replace' 

865 line : str or `ColStr` or list of str 

866 The line to make edits on. 

867 

868 Returns 

869 ------- 

870 line : str or `ColStr` or list of str 

871 The input line, but with all requested changes made. 

872 """ 

873 

874 # in case of over/under, type(line) == ColStr. 

875 # in case of left/right, type(line) == list 

876 linelen = len(line) 

877 for char, positions in instructions.items(): 

878 char = ColStr(char) 

879 charlen = len(char) 

880 # convert to the correct data type for this line 

881 if lineshort in "lr": 

882 char = list(char) 

883 

884 for pos in positions: 

885 pos[0] = pos[0] + linelen if pos[0] < 0 else pos[0] 

886 if len(pos) == 1: # if we just have starting position, get end 

887 pos.append(pos[0] + charlen) 

888 usechar = char 

889 else: 

890 pos[1] = pos[1] + linelen if pos[1] < 0 else pos[1] 

891 desgap = pos[1] - pos[0] 

892 usecharlen = charlen 

893 if desgap > charlen: 

894 usechar = char * (desgap // charlen + 1) 

895 usecharlen = charlen * (desgap // charlen + 1) 

896 if desgap < usecharlen: 

897 usechar = usechar[:desgap] 

898 

899 line = line[:pos[0]] + usechar + line[pos[1]:] 

900 return line

902 def finalize(self, color_border, color_text, padding_char, n_padding): 

903 """Brings all created elements together to form the end result 

904 

905 Parameters 

906 ---------- 

907 color_border : `ColStr` 

908 The color that the border should have 

909 color_text : `ColStr` 

910 The color that the text should have 

911 padding_char : str or `ColStr` 

912 The character used to add some space between the title 

913 text and border 

914 n_padding : int 

915 How much space should be kept (horizontally) between the 

916 text and the border 

917 """ 

918 

919 header = ColStr("") 

920 

921 # overline 

922 header += color_border + self.over 

923 if len(self.over) != 0: # If we have an upper line, add a line break 

924 header += "\n" 

925 

926 # sidelines 

927 pad = padding_char[0] * n_padding 

928 for lft, txt, rght in zip(self.left, self.title, self.right): 

929 header += ( 

930 color_border + lft + # left border 

931 color_text + pad + txt + pad + # text + padding 

932 color_border + rght + "\n") # right border 

933 

934 # underline 

935 header += color_border + self.under + GM_pt.Printer.colors.clear 

936 

937 return header
