1""" 

2Provides the command line interface. 

3 

4When a user calls GMAP, this script is activated. It makes sure the 

5correct program (GEM, Setup) is called, and performs the first 

6rudimentary parsing of command line arguments. 

7 

8Functions 

9------- 

10cmd_interface 

11 The main function orchestrating the process from GMAP command to 

12 calling the correct program. Uses the other functions in this 

13 script. 

14calldict 

15 Creates a dict out of the call command with parameter names as keys, 

16 and user choice for these as values. 

17get_safe_dark 

18 From the call dict, looks at user choices, and determines whether 

19 GMAP should run in safe mode or dark mode. 

20print_logo 

21 Prints the GMAP welcome message. Logo, version, github link, date, 

22 and command that invoked GMAP. 

23cmd_to_help 

24 Prints the correct help message. 

25cmd_to_tools 

26 Determines what (sub) program (GEM, Setup) should be called. 

27main 

28 Obtains sys.argv and calls cmd_interface. 

29"""

52def cmd_interface(callcommand): 

53 """Directs the user to the correct program. 

54 

55 Interpret what program the user would like to use, and direct the 

56 order to it. Or, when requested, print the help of GMAP or that of 

57 the specific tool instead. 

58 

59 Parameters 

60 ---------- 

61 callcommand : list of str 

62 Basically, the return value of sys.argv. What the user has 

63 actually requested from the program. 

64 """ 

65 

66 # The very first initialization the program needs/assumes. Also initializes 

67 # the printing tool (Printer). 

68 GM_fh.FileLocations(instr=callcommand) 

69 

70 # Deduce whether we should be running in safe mode / dark mode. 

71 # These 'special' flags should be caught separately to avoid issues like 

72 # 'GMAP -nodm' throwing an error. 

73 all_args = calldict(callcommand) 

74 safe_mode, dark_mode = get_safe_dark(all_args) 

75 Printer.setenv(safe_mode, dark_mode) 

76 

77 print_logo() # Print the GMAP logo + base text below 

78 

79 # get first entry form args dict -> the GMAP call (before first parameter) 

80 gmapcall, extras = next(iter(all_args.items())) 

81 call = [gmapcall] + extras 

82 

83 # yes, this looks double - want to take any length to 3, both 1 and 2. 

84 allhelps = ["help", "h", "-h"] 

85 if len(call) == 1: 

86 call.append(allhelps[0]) 

87 if len(call) == 2: 

88 call.append(allhelps[0]) 

89 

90 choice = call[1] 

91 subch = call[2] 

92 

93 if choice.lower() in allhelps: 

94 cmd_to_help(allhelps, subch) 

95 

96 elif choice in GMAP.alltools: 

97 cmd_to_tools(allhelps, callcommand, choice, subch) 

98 

99 else: 

100 _report_unknown_choice()

103def calldict(callcommand): 

104 """Create a dict off of the call command. 

105 

106 Naive logic leads the keys to be a string starting with '-', and the 

107 values to be the choices. Values can be empty lists. The first entry 

108 is different - its the program call! 

109 

110 Parameters 

111 ---------- 

112 callcommand : list of str 

113 The command used to invoke the program 

114 

115 Returns 

116 ------- 

117 all_args : dict 

118 A dictionary representation of the callcommand 

119 """ 

120 

121 # First, split args into dict. Here, the parameter name becomes the key 

122 # (except for the first one, that's program name, but dicts are ordered!) 

123 # and it's choice(s) become the value. 

124 current_arg = [callcommand[0]] 

125 all_args = {} 

126 for item in callcommand[1:]: 

127 if item.startswith("-"): 

128 # lets hope that doubles will never appear (they shouldn't, AFAIK) 

129 all_args[current_arg[0]] = current_arg[1:] 

130 current_arg = [item] 

131 else: 

132 current_arg.append(item) 

133 else: 

134 all_args[current_arg[0]] = current_arg[1:] 

135 

136 return all_args

139def get_safe_dark(all_args): 

140 """Determine whether the program has been requested to run in safe 

141 or dark mode 

142 

143 Parameters 

144 ---------- 

145 all_args : dict 

146 A dictionary representation of the callcommand 

147 

148 Returns 

149 ------- 

150 safe_mode : bool 

151 Indicates whether the program should run in safe mode (no fancy 

152 outputs, currently only disables colored output) 

153 dark_mode : bool 

154 Indicates whether the program should run in dark mode (colors 

155 chosen for a dark background) 

156 """ 

157 

158 safe_mode = False 

159 dark_mode = True 

160 # See if any urgent print-related settings are given. 

161 for parameter, choice in all_args.items(): 

162 choice_lower = [item.lower() for item in choice] 

163 

164 # If the user requests to run in safe mode 

165 if parameter.lower() in ("-safe", "--safe_mode"): 

166 # asks safe mode, and doesn't explicitly turn it off 

167 if not any(item in choice_lower for item in ("f", "false")): 

168 safe_mode = True 

169 else: 

170 safe_mode = False 

171 

172 # if the user asks to NOT use dark mode 

173 if ( 

174 ( # asks nodark, and doesn't explicity turn nodark off 

175 parameter.lower() in ("-nodm", "--nodark_mode") 

176 and not (any(item in choice_lower for item in ("f", "false"))) 

177 ) or ( # asks dark, but explicitly turns dark off 

178 parameter.lower() in ("-dm", "--dark_mode") 

179 and (any(item in choice_lower for item in ("f", "false"))) 

180 ) 

181 ): 

182 dark_mode = False 

183 elif parameter.lower() in ( 

184 "-nodm", "--nodark_mode", "-dm", "--dark_mode" 

185 ): 

186 dark_mode = True 

187 return safe_mode, dark_mode

190def print_logo(): 

191 """Prints the program logo and corresponding text. 

192 """ 

193 

194 # Now, load/print logo! 

195 with open(GM_fh.FileLocations.script_dir / "logo.txt") as lfile: 

196 logostr = lfile.read() 

197 

198 # convert abbr to actual color markers 

199 colors = Printer.colors 

200 logostr = logostr.replace("P", colors.pink_hc) 

201 logostr = logostr.replace("G", colors.green_hc) 

202 logostr += colors.clear 

203 

204 Printer.print(1, "\n\n" + logostr + "\n") 

205 Printer.print(1, f"\nRunning GMAP version {GMAP.__version__}") 

206 Printer.print( 

207 1, 

208 "\nFor the most up-to-date version of GMAP, reporting bugs/issues, " 

209 "suggesting improvements for the program, questions, or any other " 

210 "kind of feedback, go to github.com/lacourjansenlab/GMAP\n" 

211 ) 

212 

213 # print current moment 

214 Printer.print( 

215 1, 

216 f"\nCurrent date/time: {GM_fh.FileLocations.now_str}\n" 

217 ) 

218 

219 # if the program was invoked directly, we only need this one 

220 Printer.print( 

221 1, 

222 f"Running the following job:\n{GM_fh.FileLocations.instruction}\n" 

223 ) 

224 

225 # but if a different command was used for starting something python, 

226 # and this file was imported (and cmd_interface was called directly), 

227 # the actual executed instruction is not the same as the initial command. 

228 if GM_fh.FileLocations.instruction != GM_fh.FileLocations.callcommand: 

229 Printer.print( 

230 1, 

231 "Issued through the following command:\n" 

232 f"{GM_fh.FileLocations.callcommand}\n"

236def cmd_to_help(allhelps, subch): 

237 """Determines what help to print. 

238 

239 Parameters 

240 ---------- 

241 allhelps : list of str 

242 All items in this list are strings the user might use to get 

243 help to be printed. 

244 subch : str 

245 Used by user to indicate intent. 

246 """ 

247 

248 if subch.lower() in allhelps: 

249 Printer.print(0, GMAP.__doc__) 

250 Printer.quit_early() 

251 elif subch in GMAP.alltools: 

252 modch = getattr(GMAP, subch) 

253 Printer.print(0, modch.__doc__) 

254 Printer.quit_early() 

255 else: 

256 _report_unknown_choice()

259def cmd_to_tools(allhelps, callcommand, choice, subch): 

260 """Determines which tool to invoke. 

261 

262 Parameters 

263 ---------- 

264 allhelps : list of str 

265 All items in this list are strings the user might use to get 

266 help to be printed. 

267 callcommand : list of str 

268 Basically, the return value of sys.argv. What the user has 

269 actually requested from the program. 

270 choice : str 

271 Used by user to indicate intent. 

272 subch : str 

273 Used by user to indicate intent. 

274 """ 

275 

276 modch = getattr(GMAP, choice) 

277 if subch.lower() in allhelps: 

278 Printer.print(0, modch.__doc__) 

279 Printer.quit_early() 

280 else: 

281 getattr(modch, choice)(callcommand[1:])
