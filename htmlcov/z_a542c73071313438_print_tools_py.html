55class Printer(metaclass=GM_ct.Singleton): 

56 """Manages prints and logs during runtime. 

57 

58 During runtime, the program can communicate many things, but the user 

59 might not be interested in all of them. That's why verbose printing 

60 is desired. Besides that, having the prints saved in a log file is 

61 greatly desired, especially on clusters. This class allows many kinds 

62 of print functionality. 

63 

64 .. seealso:: 

65 :func:`devprint` 

66 Meant for printing during development. 

67 

68 Parameters 

69 ---------- 

70 files : :class:`~GMAP.src.tools.file_handler.FileLocations` 

71 Contains all currently known paths and other file-related properties. 

72 

73 Attributes 

74 ---------- 

75 logfile : `pathlib.path` 

76 The path to the logfile 

77 backlog : list 

78 Initally, the user-specified log file path is not yet known. 

79 Until it is, all that should be printed is instead put into the 

80 backlog. In case the program quits before the log file is known, 

81 the backlog is printed to the crash file (and terminal). 

82 program_state : str 

83 In what phase/mode the program currently exists. This state 

84 influences the desired printing behaviour. 

85 color_mode : str 

86 How colors should be printed. 24bit, 4bit, or white. 

87 colors : :class:`~GMAP.src.tools.color_schemes.PrinterColors` 

88 Colors to be used/sampled by other functions. Contains unique 

89 color names. 

90 _colors : :class:`~GMAP.src.tools.color_schemes.PrinterColors` 

91 The actual colors to be used. Contains a dict to translate the 

92 unique color names of cls.colors to ANSI escape codes. 

93 line_length : int 

94 The desired line length for both the command line and log files. 

95 dont_report_error : list of \ 

96 :class:`~GMAP.src.tools.string_classes.ErrCode` 

97 Any error codes in this list will not be reported on when 

98 encountered. Silenced fatal errors will still quit the program. 

99 timer : :class:`~Timer` 

100 The timer that keeps track of calculation times. 

101 syspathcopy : list 

102 A copy of sys.path made when files was instantiated. 

103 verbose : int 

104 How verbose the prints to the command line should be. 

105 verbose_logfile : int 

106 How verbose the prints to the log file should be. 

107 preline : list of (list of int, str pairs) or list of (list of int,\ 

108 :class: `~GMAP.src.tools.string_classes.ColStr` pairs) 

109 Short bits that should be printed at the beginning of every 

110 output line. The ints represent the verbose levels at which the 

111 companion string should be printed. These (short!) strings can 

112 either be lines/marks for pretty makeup of output, or a marker 

113 like those used for demo mode.

165 The verbosity of the message to print. This will be compared with

168 the message will be printed/logged.

188 message will be printed.

205 "instruction": instruction, 

206 "line_length": line_length, 

207 "detailed_instructions": detailed_instructions, 

208 "sep": sep} 

209 ])

238 (cls.verbose_logfile in detailed_instructions)

249 @classmethod 

250 def _print_preparation( 

251 cls, verbose, message, line_length, color_mode, wrap_preline="" 

252 ): 

253 """Format prints so they can be shown. 

254 

255 After cls.print() has decided that something should indeed be 

256 printed, this function is called. It does all the formatting 

257 and makeup of the information that should be printed: 

258 

259 - First, see if anything should (by default) come before the 

260 message to print (the preline). This could be a border, 

261 or demo-mode marker. 

262 - If so, figure out the length of this preline, and change its 

263 color to the desired print color ((2)4bit, or white) 

264 - Word wrap the message that should be printed, with the desired 

265 wrap distance adjusted for len(preline). 

266 - change the color of the message 

267 - make sure any color markers in the message are repeated (less 

268 and more automatically reset the color at linebreaks...) 

269 - add the preline to the front of each message line 

270 - return the finalized message. 

271 

272 Parameters 

273 ---------- 

274 verbose : int 

275 The verbose level at which the message will be printed 

276 message : str or :class:`GMAP.src.tools.string_classes.ColStr` 

277 The message to format for printing 

278 line_length : int 

279 How many characters can at most be on a line. 

280 color_mode : str 

281 What color palette should be used. Can be '24bit', '4bit' or 

282 'white' (colorless). 

283 wrap_preline : str 

284 When wrapping the line, this bit should be added by the 

285 wrapper to make things line up nicely. 

286 

287 Returns 

288 ------- 

289 outmessage : same type as message 

290 The input message, fully formatted. 

291 """ 

292 

293 # The color-repeat tool is much, much easier to write/create if the 

294 # input color format is fixed! So, that's why that step is earlier in 

295 # the process. 

296 

297 # Figure out (the length of) printpreline, and change its color to 

298 # the desired color (so they can be returned directly). 

299 printpreline = GM_sc.ColStr("").join( 

300 [item[1] for item in cls.preline if verbose in item[0]]) 

301 printpreline = printpreline.change_color(color_mode) 

302 prelinelen = len(printpreline) 

303 

304 # wordwrap the message to be printed (adjusted length for preline) 

305 message = GM_sc.ColStr(message).wrap( 

306 line_length - prelinelen, wrap_preline) 

307 # change the message color (so color_repeater knows what to expect) 

308 # make sure color markers/codes are repeated after a line break 

309 message = message.change_color(color_mode).repeat_color() 

310 

311 # add prelines 

312 message = printpreline + message.replace("\n", f"\n{printpreline}") 

313 

314 return message

353 """Warning system. Prints the message, and allows to force-quit after. 

354 

355 Parameters 

356 ---------- 

357 message : str 

358 The warning/error message to print. 

359 error_code : str 

360 The error code belonging to the warning that is raised - to be 

361 reported to the user. See the 

362 :ref:`warning overview<UserGuide_page_warning_overview>` page. 

363 exitbool : bool, default=False 

364 Whether the program should stop due to this error. 

365 exception : BaseException, default=None 

366 If the warning corresponds to a 'basic' python error, 

367 that error can be caught and fed into this function. 

368 GMAPerrclass : BaseException, default=None 

369 The exact exception that should be raised if the error is 

370 fatal. One should pick one from GMAP.src.tools.exceptions. 

371 """ 

372 

373 # if this default is set directly in the function signature, a circular 

374 # reference problem occurs, and this module MUST be imported before 

375 # the exceptions module is imported. This way, the import order does 

376 # not matter. 

377 if GMAPerrclass is None: 

378 GMAPerrclass = GM_ex.GMAPexception 

379 

380 if not message.startswith("\n"): 

381 message = "\n" + message 

382 

383 # might seem backwards, but we should report if the error wasn't 

384 # silenced. 

385 if GM_sc.ErrCode(error_code) not in cls.dont_report_error: 

386 if exitbool: 

387 printinstruct = "f" 

388 else: 

389 message = ( 

390 f"\n{cls.colors.red_hc}WARNING:" + cls.colors.red_todef 

391 + message + cls.colors.clear)

393 

394 error_message = message 

395 cls.print(0, message, instruction=printinstruct) 

396 

397 # print the traceback in exactly the same way as it would be 

398 # thrown into the command line. 

399 if exception: 

400 traceprint = TbEx.from_exception(exception).format() 

401 msg = ( 

402 "\n" + cls.colors.red_todef + "".join(traceprint) 

403 + cls.colors.clear) 

404 cls.print(4, msg, instruction=printinstruct) 

405 if cls.verbose == 4: 

406 error_message += msg 

407 

408 msg = ( 

409 "More information can be found in the documentation " 

410 "user pages using the following error " 

411 f"code: {cls.colors.clear} {error_code}" 

412 ) 

413 cls.print( 

414 0, cls.colors.red_todef + msg, 

415 instruction=printinstruct) 

416 error_message += msg 

417 

418 if exitbool: 

419 if cls.backlog: 

420 cls.print_backlog() 

421 # We want an empty line before the error 

422 cls.print(0, f"\n{cls.colors.red_hc}", instruction="p", end="") 

423 error_message = (

537class Timer: 

538 """Manages the timekeeping during runtime. 

539 

540 Times can be either added, or read from here. 

541 

542 Parameters 

543 ---------- 

544 start : int, default=None 

545 If provided, this is used as the reference time, instead of the 

546 time at which the class was created. This new starting time 

547 should be created using time.perf_counter_ns() 

548 

549 Attributes 

550 ---------- 

551 zero : int 

552 The reference point to which all times should be compared. 

553 times : dict of str: int pairs 

554 The different times that the timer was requested to save. The 

555 keys are the messages the times were accompanied by, the values 

556 are the actual (raw perf_counter_ns()) times. 

557 totals : dict of str: int pairs 

558 The cumulative times the timer was requested to save. Allow to 

559 see (over many frames) how long the program spent where. 

560 previous : list of str, int 

561 What the previous time stamp was. If we call the current, only 

562 then do we know how long we spent in the previous step (and we 

563 can save it to totals). 

564 

565 Notes 

566 ----- 

567 All labels in use (refers to what'll be done next):: 

568 

569 (start) 

570 ParParse (3) 

571 AddMaps (3) 

572 MDinit (2) 

573 MapInit (3) 

574 ClibLoad (3) 

575 PrepLoop (3) 

576 StartLoop (3) 

577 FrameUpdate (4) 

578 PosBox (4) 

579 COM (4) 

580 OscUpdate (4) 

581 StructInit (4) 

582 MapFInit (4) 

583 Calc (4) 

584 VEGprop (4) 

585 VEGcalc (5) 

586 VEGuse (5) 

587 PrepCoup (4) 

588 CalcCoup (4) 

589 MapFPost (4) 

590 FrameWrite (4) 

591 LoadFrame (4) 

592 MapPost (3) 

593 """ 

594 

595 def __init__(self, start=None): 

596 if start is None: 

597 self.zero = time.perf_counter_ns() 

598 else: 

599 self.zero = start 

600 

601 self.times = {} 

602 self.totals = {} 

603 self.previous = ["start", self.zero] 

604 

605 def add_time(self, msg): 

606 """Add another time to the dict. 

607 

608 Parameters 

609 ---------- 

610 msg : str 

611 The key with which the time is stored. 

612 """ 

613 

614 now = time.perf_counter_ns() 

615 

616 if self.previous[0] not in self.totals: 

617 self.totals[self.previous[0]] = now - self.previous[1] 

618 else: 

619 self.totals[self.previous[0]] += now - self.previous[1] 

620 self.times[msg] = now 

621 self.previous = [msg, now] 

622 

623 def get_time(self, msg): 

624 """Retrieve a time from the dict. 

625 

626 Times retrieved have self.zero subtracted first, so they become 

627 useful/meaningful. 

628 

629 Parameters 

630 ---------- 

631 msg : str 

632 The key from which the time should be retrieved. 

633 """ 

634 

635 return self.times[msg] - self.zero 

636 

637 def get_total_ns(self, *args): 

638 """Get sum of total calculation times. 

639 

640 Each label listed in args is retrieved from totals, which is 

641 subsequently summed. 

642 

643 Parameters 

644 ---------- 

645 *args : str 

646 Each argument is a string and represents a label of which 

647 the time should be retrieved. 

648 """ 

649 

650 total_time = 0 

651 for label in args: 

652 total_time += self.totals[label] 

653 return total_time 

654 

655 def get_total_format(self, *args): 

656 """Get a nicely formatted representation of total calculation times. 

657 

658 Each label listed in args is retrieved from totals, and they 

659 are summed before conversion to a human-readable format. 

660 

661 Parameters 

662 ---------- 

663 *args : str 

664 Each argument is a string and represents a label of which 

665 the time should be retrieved. 

666 """ 

667 

668 total_time = self.get_total_ns(*args) 

669 return time_to_str(total_time)

679 :func:`GMAP.src.tools.plotter.plot_color_conv`. 

680 

681 Only when pandas is an issue, or when the commandline

690 those last lines), and adjust the `range_` definition to start at

693

696 for r in range_: 

697 Printer.print( 

698 0, f"\nRed value: {r}", instruction="p", line_length=130)

710 string += " "

712 if r == 128: 

713 break

716def word_wrap(string, deslen=79, wrap_preline=""): 

717 """Formats a given string to create soft-wrap-like behaviour. 

718 

719 Parameters 

720 ---------- 

721 string : str or :class:`~GMAP.src.tools.string_classes.ColStr` 

722 The string to format. 

723 deslen : int, default=79 

724 The maximum amount of characters per line. 

725 

726 Returns 

727 ------- 

728 new_string : str or :class:`~GMAP.src.tools.string_classes.ColStr` 

729 The original input `string`, but with newline characters added 

730 where necessary. Retains input type. 

731 wrap_preline : str 

732 When wrapping the line, this bit should be added by the 

733 wrapper to make things line up nicely. 

734 """ 

735 

736 intype = type(string) 

737 colstr = GM_sc.ColStr(string) 

738 startlst = colstr.split("\n") 

739 endlst = [] 

740 

741 # We always take the length of the 'white' string, not the original; 

742 # color swaps don't take up space in the command line, but here they do 

743 # represent a length of up to 16! 

744 

745 # We will split at whitespaces (words), but we do want to print all white 

746 # spaces (except for those replaced by a line break). Therefore, when 

747 # adding an item, the whitespace is added back in. However, for the very 

748 # first item, that shouldn't happen. Therefore, we make sure to omit that 

749 # very first one 

750 

751 for item in startlst: # each item is of type ColStr 

752 startline = True # Remember: we are now at the start of a line 

753 ilen = len(item) # ColStrs take their colors into account for length 

754 if ilen > deslen: 

755 itemlst = item.split(" ") 

756 buildstr = "" 

757 for subitem in itemlst: 

758 slen = len(subitem) 

759 blen = len(buildstr) 

760 if blen + slen >= deslen: 

761 # if this is the first of the line, that line starts with 

762 # an extra whitespace that should be stripped 

763 if startline: 

764 endlst.append(buildstr[1:]) 

765 startline = False # after this, no longer first 

766 else: 

767 endlst.append(buildstr) 

768 buildstr = wrap_preline + subitem 

769 else: 

770 buildstr += " " + subitem 

771 endlst.append(buildstr) 

772 else: 

773 endlst.append(item) 

774 

775 return intype("\n").join(endlst)

778def devprint(*args, **kwargs): 

779 """python print for developers 

780 

781 Developers should use this function to print instead of the python 

782 print function - it also prints where it was evoked, so any random 

783 prints can be easily retraced and removed after they're no longer 

784 useful. 

785 

786 .. note:: 

787 Calling this function goes **exactly** like the print from 

788 python 

789 

790 Parameters 

791 ---------- 

792 *args : tuple 

793 All the things to print 

794 **kwargs : tuple 

795 The keyword arguments to pass to python print. 

796 """

801 filename = pathlib.Path(cf_i.filename).name 

802 funcname = cf_i.function

804 f"(line {lineno:4d})", 

805 *args, 

806 f"(from {funcname} in {filename})", 

807 **kwargs 

808 )

811def header( 

812 verbose, title, preset, preline=None, newlines=None, **kwargs 

813): 

814 """Create a header from the given text. 

815 

816 A call to footer with the exact same parameters will generate the 

817 corresponding closing structure. The header is directly printed, 

818 not returned. 

819 

820 Parameters 

821 ---------- 

822 verbose : int 

823 The verbose level on which the header should be printed. Is 

824 directly passed to the :func:`Printer.print` call. 

825 tile : str 

826 The text that should be within the header 

827 preset : str 

828 What style header should be used. Current options: 

829 

830 custom : 

831 The caller fully decides. Any additional kwargs to this 

832 function are passed to :func:`_make_header`. See the 

833 documentation of that function for available kwargs. 

834 newlines default = (0, 0) 

835 nohead : 

836 No special header is made - title is taken as-is. 

837 doublebox : 

838 The header receives a border of the ascii double-line box 

839 character set, with a double-line dropping down to mark all 

840 output belonging to the section:: 

841 ╔═════════════╗ 

842 ║ input_title ║ 

843 ╚╦════════════╝ 

844 ║ text with the new set preline 

845 

846 The matching footer should be used too. 

847 Newlines default = (2, 1) 

848 

849 doublebox_bare : 

850 The header receives a border of the ascii double-line box 

851 character set. There is no line dropped:: 

852 ╔═════════════╗ 

853 ║ input_title ║ 

854 ╚═════════════╝ 

855 text will be printed with no set preline 

856 

857 No footer is required. 

858 Newlines default = (2, 1) 

859 preline : str, default=None 

860 Only used if preset set to "custom" or "nohead". If the text 

861 following the header should have a preline, this will be 

862 arranged. 

863 newlines : tuple of two ints, default=None 

864 The amount of empty lines that should be printed before and 

865 after the header, respectively. If set to None, the 

866 preset-specific default will be used. 

867 **kwargs : any 

868 Any kwargs that should be passed to either 

869 :func:`Printer.print` or :func:`_make_header`. 

870 """ 

871 

872 if "color_border" not in kwargs or kwargs["color_border"] is None: 

873 cb = Printer.colors.green_lc 

874 

875 if "color_text" not in kwargs or kwargs["color_text"] is None: 

876 _ = Printer.colors.clear 

877 

878 cc = Printer.colors.clear 

879 

880 if newlines is None: 

881 match preset: 

882 case "nohead" | "custom": 

883 newlines = (0, 0) 

884 case "doublebox" | "doublebox_bare": 

885 newlines = (2, 1) 

886 

887 head = "\n" * newlines[0] 

888 match preset: 

889 case "nohead" | "custom": 

890 if preset == "custom": 

891 title = GM_sc.Header(title, **kwargs).s 

892 head += title 

893 Printer.print(verbose, head, **kwargs) 

894 if preline is not None: 

895 Printer.preline.append([[*range(verbose, 5)], preline]) 

896 case "doublebox" | "doublebox_bare": 

897 if preset == "doublebox": 

898 special = {"u": {"replace": {"╦": [[1]]}}} 

899 else: 

900 special = None 

901 head += GM_sc.Header( 

902 title, linemode="oulrc", padding=1, overline_char="═", 

903 underline_char="═", left_char="║", right_char="║", 

904 corner_char="╔╗╚╝", special=special 

905 ).s 

906 Printer.print(verbose, head, **kwargs) 

907 if preset != "doublebox_bare": 

908 Printer.preline.append([ 

909 [*range(verbose, 5)], cb + " ║ " + cc]) 

910 if newlines[1] > 0: 

911 Printer.print(verbose, "\n" * (newlines[1] - 1), **kwargs)

1004def intlist_to_rangelist(intlist, n_int, make_shadow=True): 

1005 """Takes a list of integers and packs it into ranges. 

1006 

1007 Parameters 

1008 ---------- 

1009 intlist : list of int 

1010 The list of integers to be packed. 

1011 n_int : int 

1012 The amount of integers that can at most be there. (The provided 

1013 value itself will never appear!) 

1014 make_shadow : bool, default=True 

1015 Whether the opposite should also be built - a list of all 

1016 indices that weren't in the intlist 

1017 

1018 Returns 

1019 ------- 

1020 result : list of str 

1021 All indices (grouped in ranges) that were in intlist. 

1022 shadow : list of str or None 

1023 All indices (grouped in ranges) that weren't in intlist. 

1024 

1025 Examples 

1026 -------- 

1027 Groups of 3 integers or larger will be grouped 

1028 

1029 >>> intlist = [*range(10)] 

1030 >>> intlist.remove(5) 

1031 >>> intlist_to_rangelist(intlist, 10) 

1032 (["0-4", "6-9"], ["5"]) 

1033 

1034 Groups of 2 integers are left as is 

1035 

1036 >>> intlist = [0, 1, 5, 8, 9, 10] 

1037 >>> intlist_to_rangelist(intlist, 11) 

1038 (["0", "1", "5", "8-10"], ["2-4", "6", "7"]) 

1039 """ 

1040 

1041 result = [] 

1042 prevnum = intlist[0] # loop will go over [1:] 

1043 range_start = intlist[0] 

1044 # so we don't need for-else to deal with last item 

1045 intlist = intlist + [None] 

1046 

1047 if make_shadow: 

1048 if prevnum != 0: 

1049 shadow = rangestrlist(0, prevnum) # all numbers not in intlist 

1050 else: 

1051 shadow = [] 

1052 else: 

1053 shadow = None 

1054 

1055 for num in intlist[1:]: 

1056 # if we don't logically count to the next one 

1057 if num != prevnum + 1: 

1058 

1059 # put missing numbers in the shadow list 

1060 if make_shadow: 

1061 if num is None: 

1062 end = n_int 

1063 else: 

1064 end = num 

1065 if prevnum + 1 != end: 

1066 shadow += rangestrlist(prevnum + 1, end - 1) 

1067 

1068 # the last number was 'alone' 

1069 result += rangestrlist(range_start, prevnum) 

1070 

1071 range_start = num 

1072 

1073 prevnum = num 

1074 

1075 return result, shadow

1078def rangestrlist(start, stop): 

1079 """Given a start and stop, return separate items or range. 

1080 Inclusive. 

1081 

1082 Meant as a helper function for :func"`intlist_to_rangelist`, not 

1083 intended for separate use. 

1084 

1085 Parameters 

1086 ---------- 

1087 start : int 

1088 The first int to be included. 

1089 stop : int 

1090 The last int to be included. 

1091 

1092 Returns 

1093 ------- 

1094 out : list of str 

1095 The grouped ints. 

1096 

1097 Examples 

1098 -------- 

1099 Groups are shortened: 

1100 

1101 >>> rangestrlist(3, 60) 

1102 ["3-60"] 

1103 

1104 >>> rangestrlist(3, 5) 

1105 ["3-5"] 

1106 

1107 Pairs are left as-is 

1108 

1109 >>> rangestrlist(3, 4) 

1110 ["3", "4"] 

1111 

1112 Single values are left alone 

1113 

1114 >>> rangestrlist(3, 3) 

1115 ["3"] 

1116 """ 

1117 if start == stop: 

1118 return [str(start)] 

1119 elif start == stop - 1: 

1120 return [str(start), str(stop)] 

1121 else: 

1122 return [f"{start}-{stop}"]

1125def time_to_str(ns_time, precision="s"): 

1126 """Converts an amount of ns into a formatted string. 

1127 

1128 Parameters 

1129 ---------- 

1130 ns_time : int 

1131 An amount of nanoseconds 

1132 precision : str, default=s 

1133 To what precision the string should be printed. 's' for seconds, 

1134 'ms' for milliseconds, 'us' for microseconds or 'ns' for 

1135 nanoseconds. 

1136 

1137 Returns 

1138 ------- 

1139 str_time : str 

1140 The input time formatted into a string. 

1141 """ 

1142 

1143 ns = ns_time % 1000 

1144 remainder = ns_time // 1000 # this is now in units of us 

1145 

1146 us = remainder % 1000 

1147 remainder //= 1000 # this is now in units of ms 

1148 

1149 ms = remainder % 1000 

1150 remainder //= 1000 # this is now in units of s 

1151 

1152 s = remainder % 60 

1153 remainder //= 60 # this is now in units of m 

1154 

1155 m = remainder % 60 

1156 remainder //= 60 # this is now in units of h 

1157 

1158 h = remainder % 24 

1159 d = remainder // 24 

1160 

1161 str_time = f"{d}-{h:02d}:{m:02d}:{s:02d}" 

1162 

1163 if precision in ("ms", "us", "ns"): 

1164 str_time += f".{ms:03d}" 

1165 

1166 if precision in {"us", "ns"}: 

1167 str_time += f"{us:03d}" 

1168 

1169 if precision == "ns": 

1170 str_time += f"{ns:03d}" 

1171 

1172 return str_time
