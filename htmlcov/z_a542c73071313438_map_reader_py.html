64class Map: 

65 """Contains all information regarding a single map. Base to build 

66 upon. 

67 

68 In this context, a map is as defined in the computational 

69 spectroscopy community - a way of estimating the properties of a 

70 functional group. It is not a python object. 

71 

72 .. note :: 

73 Users of the program are probably looking for the 

74 :ref:`adding a new map<UserGuide_page_adding_map>` page. 

75 

76 .. important ::

78 Multiple methods live here to avoid code duplication, but are 

79 not created by the class itself and require attributes not 

80 present/instantiated here either.

83 following:

159 """Creates a RefPars object for the map-specific parameters

168 stored as the `self.ref_pars` attribute.

179 def find_rawpars(self, cmd_pars, in_pars, def_pars): 

180 """Creates CmdPars, InPars and DefPars objects for this map instance. 

181 

182 Searches through the provided cmd_pars, in_pars and def_pars to see 

183 whether there are any map-type parameters belonging to this map. 

184 If so, they are taken from there, put in the map-specific 

185 instances for cmd_pars, in_pars and def_pars, and then removed from 

186 the source (as the source will be checked for emptiness at the 

187 end). 

188 

189 Parameters 

190 ---------- 

191 cmd_pars : :class:`~GMAP.src.tools.parameter_parser.RawPars` 

192 The object storing all parameters provided on the command 

193 line. 

194 in_pars : :class:`~GMAP.src.tools.parameter_parser.RawPars` 

195 The object storing all parameters provided in the input 

196 parameter file. 

197 def_pars : :class:`~GMAP.src.tools.parameter_parser.RawPars` 

198 The object storing all parameters provided in the default 

199 parameter file. If no such file was provided, ref_pars is 

200 used instead. 

201 """ 

202 

203 # for each parameter source, extract all choices belonging to this 

204 # map, and make a RawPars object with those choices. Make sure 

205 # to empty the not_found array, so each source can be checked to make 

206 # sure all parameters are understood. 

207 

208 # first, cmd_pars 

209 map_pars = self.extract_notfound(cmd_pars) 

210 self.cmd_pars = GM_pp.RawPars.from_dict( 

211 "Cmdline", map_pars, self.ref_pars, False 

212 ) 

213 for parname in map_pars.keys(): 

214 del cmd_pars.not_found[self.name + "." + parname] 

215 

216 # in_pars 

217 map_pars = self.extract_notfound(in_pars) 

218 self.in_pars = GM_pp.RawPars.from_dict( 

219 in_pars.fname, map_pars, self.ref_pars, False 

220 ) 

221 for parname in map_pars.keys(): 

222 del in_pars.not_found[self.name + "." + parname] 

223 

224 # def_pars 

225 if type(def_pars) is GM_pp.RawPars: 

226 map_pars = self.extract_notfound(def_pars) 

227 self.def_pars = GM_pp.RawPars.from_dict( 

228 def_pars.fname, map_pars, self.ref_pars, True 

229 ) 

230 for parname in map_pars.keys(): 

231 del def_pars.not_found[self.name + "." + parname] 

232 else: 

233 self.def_pars = GM_pp.RawPars.create_empty()

235 def extract_notfound(self, raw_par_inst): 

236 """Find all parameters of this map in the given RawPars 

237 instance. 

238 

239 Parameters 

240 ---------- 

241 RawParInst : :class:`~GMAP.src.tools.parameter_parser.RawPars` 

242 The instance through which to look for parameters that 

243 belong to this map. 

244 

245 Returns 

246 ------- 

247 map_pars : dict 

248 The 'slice' of the RawParInst.not_found dict that contains 

249 parameters starting with 'self.name'. 

250 """ 

251 map_pars = {} 

252 for parname, choice in raw_par_inst.not_found.items(): 

253 parnamelist = parname.split(".") 

254 if len(parnamelist) != 2: 

255 GM_pt.Printer.warning( 

256 "\nNames of map-specific parameters cannot contain a '.'.", 

257 "SU_MR_1", True, GMAPerrclass=GM_ex.GmapValueError 

258 ) 

259 if parnamelist[0] != self.name: 

260 continue 

261 map_pars[parnamelist[1]] = choice 

262 

263 return map_pars

285 def extract_code(self): 

286 """Imports the main.py file and returns its module instance. 

287 

288 Taking ``import numpy as np`` as example, ``np`` is the module 

289 instance. 

290 

291 Returns 

292 ------- 

293 module : module 

294 The contents of the file as a module. 

295 """ 

296 

297 modpath = self.directory / "main.py" 

298 if not modpath.is_file(): 

299 return None 

300 

301 modname = self.name + "_mainpy_code" 

302 

303 try: 

304 spec = importlib.util.spec_from_file_location(modname, modpath) 

305 module = importlib.util.module_from_spec(spec) 

306 sys.modules[modname] = module 

307 sys.path.insert(1, str(self.directory)) 

308 spec.loader.exec_module(module) 

309 except Exception as ex: 

310 GM_pt.Printer.warning( 

311 "\nA problem occured while reading in the code for the map " 

312 f"{self.name}, defined at " 

313 f"{str(self.directory.resolve())}. " 

314 "Please consult the information of this map, or contact the " 

315 "developer of this map.", 

316 "MI_MR_1", False, ex 

317 ) 

318 return None 

319 return module

321 def complete_code(self, funcnames, kwargslist=None): 

322 r"""Adds missing functions to the code of this map. 

323 

324 A map can have many different funtions called by the program to 

325 allow for fully custom behaviour. To make it easier to call them 

326 later, any that do not exist yet, will have a default function 

327 added in. For some, that may be a useless shell (just pass), 

328 for others, there will be an actual default calculation. 

329 

330 Parameters 

331 ---------- 

332 funcnames : tuple of str 

333 The functions that should be added in (without the GM\_ 

334 prefix) 

335 kwargslist : tuple of dicts or NoneType, default=None 

336 If any of the requested functions (in funcnames) require 

337 arguments, they should be supplied as kwargs in a dict. 

338 If any function needs them, the kwargs for all must be 

339 provided, this tuple must have the same length as funcnames. 

340 

341 Note 

342 ---- 

343 The `code` attribute is not actually created by this class. It 

344 assumes subclasses create it, before calling this function to 

345 update them. This function is also not called by any function 

346 in this class. 

347 """ 

348 

349 if not kwargslist: 

350 kwargslist = [{}] * len(funcnames) 

351 

352 for funcname, kwargs in zip(funcnames, kwargslist): 

353 if not hasattr(self.code, "GM_" + funcname): 

354 setattr( 

355 self.code, "GM_" + funcname, 

356 getattr(GM_dmf, "get_" + funcname)(**kwargs) 

357 )

359 def append_core(self): 

360 """Interprets the choice in core.txt for add_corefile. 

361 

362 Appends the contents of the requested file(s). If added files 

363 rely on files themselves, those dependencies are also added. 

364 

365 Note 

366 ---- 

367 The `rawcore` attribute is not actually created by this class. 

368 It assumes subclasses create it, before calling this function to 

369 update them. This function is also not called by any function 

370 in this class. 

371 """ 

372 

373 all_to_add = [ 

374 file for files in self.rawcore.get("add_corefile", []) 

375 for file in files 

376 ] 

377 self.rawcore["add_corefile"] = [] 

378 # as long as things should be added - this allows imports to rely 

379 # on imports themselves. 

380 while all_to_add: 

381 for name in all_to_add: 

382 # we only want the first occurence of this file 

383 fpaths = [ 

384 file for file in self.avail_files if file.name == name 

385 ] 

386 

387 try: 

388 self.rawcore = self.parse_core( 

389 fpaths[0], self.rawcore 

390 ) 

391 if not self.rawcore: 

392 self.success = False 

393 return 

394 except IndexError as ex: 

395 GM_pt.Printer.warning( 

396 f"\nA file of the name {name} was requested to be " 

397 "added " 

398 f"to the file {self.corepath}. However, no file of " 

399 "that name could be found", 

400 "MI_MR_6", False, exception=ex 

401 ) 

402 self.success = False 

403 return 

404 except Exception as ex: 

405 GM_pt.Printer.warning( 

406 f"\nThe file {fpaths[0]} was requested to be added to " 

407 f"the file {self.corepath}. However, an issue " 

408 "occured, so it cannot be added.", 

409 "MI_MR_5", False, exception=ex 

410 ) 

411 self.success = False 

412 return 

413 all_to_add = [ 

414 file for files in self.rawcore.get("add_corefile", []) 

415 for file in files 

416 ] 

417 self.rawcore["add_corefile"] = []

419 def find_core(self): 

420 """Sees if the map core exists, and extracts all its information. 

421 

422 The validity is not confirmed in any way - that will come later. 

423 First, just extracting it, so that the GM_adjust_map_core_raw 

424 function of the map may change it. 

425 

426 The returned dict contains the first word of a line as the key, 

427 and the rest of the line split into a list as the value. 

428 Parameters that are allowed to occur on multiple lines have 

429 a list of all lines split into lists as their value instead. 

430 

431 Returns 

432 ------- 

433 core_contents : dict of str - list of str pairs 

434 The contents of the file, not yet analyzed for validity. 

435 """ 

436 

437 filepath = (self.directory / "core.txt").resolve() 

438 if not filepath.is_file(): 

439 # This file is mandatory for singles maps 

440 if isinstance(self, SingleMap): 

441 GM_pt.Printer.warning( 

442 f"\nCould not find the file {filepath}. This file is " 

443 "required for the map to function. Please consult the " 

444 "information of this map, or contact the developer of " 

445 "this map.", 

446 "MI_MR_2", False 

447 ) 

448 self.success = False 

449 return None 

450 

451 # This file is optional for pairs maps 

452 elif isinstance(self, PairMap): 

453 return {} 

454 

455 if not GM_fh.check_file_readability(filepath, doquit=False): 

456 self.success = False 

457 return None 

458 

459 self.corepath = filepath 

460 

461 core_contents = self.parse_core(filepath) 

462 if not core_contents: 

463 self.success = False 

464 return None 

465 

466 return core_contents

468 @staticmethod 

469 def parse_core(filepath, file_contents=None): 

470 """Actually retrieves the information from core.txt. 

471 

472 Parameters 

473 ---------- 

474 filepath : pathlib.Path 

475 The location at which the file is stored. 

476 file_contents : dict or NoneType, default=None 

477 Any contents belonging to core.txt that have already been 

478 found. This is applicable when using the 'append_core' 

479 keyword, which in spirit concatenates file contents. 

480 

481 Returns 

482 ------- 

483 core_contents : dict of str - list of str pairs 

484 The contents of the file, not yet analyzed for validity. 

485 """ 

486 

487 if file_contents is None: 

488 file_contents = {} 

489 

490 with open(filepath) as fhand: 

491 for line in fhand: 

492 line = SingleMap.cleanline(line) 

493 if line: 

494 file_contents = SingleMap.parse_core_line( 

495 line, file_contents, filepath 

496 ) 

497 if not file_contents: 

498 return None 

499 return file_contents

501 @staticmethod 

502 def cleanline(line): 

503 """Cleans up a line from core.txt for parsing 

504 

505 If there is a '#' on the line, ignore it and everything after 

506 it. Interpret multiple whitespaces back-to-back as a single one. 

507 

508 Parameters 

509 ---------- 

510 line : str 

511 The line as directly read from the file. 

512 

513 Returns 

514 ------- 

515 line : list of str 

516 The 'words' on the line. Each item is anything but 

517 whitespace. 

518 """ 

519 

520 line = GM_pp.cleanline(line) 

521 line = line.strip().split() 

522 line = [x for x in line if x] 

523 return line

525 @staticmethod 

526 def parse_core_line(line, file_contents, filepath): 

527 """Parses a single line from core.txt 

528 

529 Parameters 

530 ---------- 

531 line : list of str 

532 A cleaned version of the line. Already split into a list, 

533 all whitespaces removed. 

534 file_contents : dict of str - list of str pairs 

535 The contents of the file, not yet analyzed for validity. 

536 Here, it is still incomplete, being built by this function 

537 each iteration. 

538 filepath : pathlib.Path 

539 The location at which the file is stored. 

540 

541 Returns 

542 ------- 

543 file_contents : dict of str - list of str pairs 

544 The contents of the file, not yet analyzed for validity. 

545 Here, it is still incomplete, being built by this function 

546 each iteration. 

547 """ 

548 

549 keyword = line[0] 

550 choice = line[1:] 

551 

552 if not choice: 

553 GM_pt.Printer.warning( 

554 f"\nNo choice detected for parameter {keyword} in the file " 

555 f"{filepath}. This issue must be fixed before this map can " 

556 "be used.", 

557 "MI_MR_3", False 

558 ) 

559 return None 

560 if keyword in ( 

561 "functional_group", "add_corefile", "influencer_group", 

562 "valid_combinations" 

563 ): 

564 if keyword in file_contents: 

565 file_contents[keyword].append(choice) 

566 else: 

567 file_contents[keyword] = [choice] 

568 return file_contents 

569 elif keyword not in file_contents: 

570 file_contents[keyword] = choice 

571 return file_contents 

572 else: 

573 GM_pt.Printer.warning( 

574 f"\nThe parameter {keyword} appeared more than once in " 

575 "the file " 

576 f"{filepath}. It may only occur once. This issue must be " 

577 "fixed before this map can be used.", 

578 "MI_MR_4", False 

579 ) 

580 return None

592class SingleMap(Map): 

593 """A :class:`Map` subclass for Singles maps. 

594 

595 Singles maps are used for computing information that only requires a 

596 single oscillator taking part (like absorbing frequency, dipole 

597 moment). 

598 

599 Attributes 

600 ---------- 

601 code : module 

602 The code that goes along with this map, if present. If not, an 

603 empty module will be filled with default functions. If it is 

604 present, any missing functions will be added from the default 

605 functions. 

606 rawcore : dict of str: list of str pairs 

607 The contents of core.txt, after very simple parsing 

608 corepath : `pathlib.Path` 

609 The path to/of the core.txt file. Dynamically created upon 

610 calling `find_core` when `initialize` is called. This function 

611 should be called in the program flow after all GMAP parameters 

612 are initialized, and before loading in MD data. 

613 core : :class:`Core` 

614 The contents of core.txt, fully parsed and ready to use. 

615 """ 

616 

617 def initialize(self): 

618 """Initializes the map. 

619 

620 Initializing is a multi-step process: 

621 

622 - interpret reference file if present 

623 - If there is a main.py file, read/extract it. 

624 - If any of GM_adjust_[run_pars/map_core_raw/oscillators] are 

625 missing, add the default for them. 

626 - Run GM_adjust_run_pars 

627 - Find the core.txt file, parse to rawcore. Supplement any 

628 files, if requested. 

629 - Run GM_adjust_map_core_raw 

630 - Parse the final choice of rawcore to core 

631 - If not present in self.code, create functions for 

632 GM_calculate_dipole and GM_get_rotation matrix based on core. 

633 """ 

634 

635 self.parse_references() 

636 

637 self.code = self.extract_code() 

638 if not self.code: 

639 self.code = GM_dmf.NewModule() 

640 

641 # not all code can be added in yet - we need to make sure that 

642 # we have adjust_run_pars and adjust_map_core_raw, as they might 

643 # influence core.txt in ways that influence the functions 

644 # defined later. 

645 self.complete_code(( 

646 "adjust_run_pars", 

647 "adjust_map_core_raw", 

648 "adjust_oscillators", 

649 "filter_oscillators" 

650 )) 

651 self.code.GM_adjust_run_pars(self) 

652 

653 # simple parse of core.txt 

654 self.rawcore = self.find_core() 

655 if not self.success: 

656 return 

657 

658 # add extra corefiles to the main core 

659 self.append_core() 

660 if not self.success: 

661 return 

662 

663 # allow the contents of core.txt to be changed 

664 self.code.GM_adjust_map_core_raw(self) 

665 

666 self.core = SingleCore(self) 

667 if not self.core.success: 

668 self.success = False 

669 return 

670 

671 # adding GM_get_dipole_dir and GM_get_rotation_matrix to self.code. 

672 self.code_add_builds() 

673 if not self.success: 

674 return 

675 

676 # make sure that all functions that must be present, are. 

677 self.assert_function_presence() 

678 

679 # adding more core-dependent functions to self.code. 

680 if self.core.electrostatic_choice: 

681 self.complete_code(("get_VEG_ref",), ({"map_": self},)) 

682 if self.core.can_output.intersection(["ham", "ene", "dip"]): 

683 self.complete_code(("calculate_dipole",), ({"map_": self},))

1523 @staticmethod 

1524 def funcgroup_parse_infile(funcgroup): 

1525 """Parse the choice made for functional_groups 

1526 

1527 This is the core of the parser. After catching blatant issues 

1528 and enforcing a single format independent of source, this 

1529 function is the next step in the process to store whats on these 

1530 lines. 

1531 

1532 output formats: 

1533 

1534 - all_opts is a list(1). 

1535 

1536 - Each item in list(1) corresponds to a single line/newstruct 

1537 (depends of source) and is itself a list(2). 

1538 - Each item in list(2) corresponds to a single residue, and is 

1539 itself a list of 2 items. The first item contains the 

1540 residue name (is a list of str), the second item contains 

1541 the atom names. The second item is itself a list(3). 

1542 - Each item in list(3) corresponds to an atom and is itself a 

1543 list of str. 

1544 

1545 - all_bonds is a list(1). 

1546 

1547 - Each item in list(1) corresponds to a single line/newstruct 

1548 (depends of source) and is itself a list(2). 

1549 - Each item in list(2) corresponds to a bond and is a list of 

1550 two items. These are the indices of the two atoms that form 

1551 the bond. 

1552 

1553 Parameters 

1554 ---------- 

1555 funcgroup : list of list of str 

1556 The choice for functional_group as stored in rawcore. 

1557 

1558 Returns 

1559 ------- 

1560 all_opts : list of list of list of list of list of str 

1561 The more structured version of what functional group(s) this 

1562 map operates on. See explanation above for more details 

1563 about the datatype. 

1564 all_bonds : list of list of list of int 

1565 All bonds that are indicative of the functional group(s) 

1566 this map operates on. This list doesn't have to be complete, 

1567 these bonds are those that make the difference. If there 

1568 are multiple residues in a single functional group, there 

1569 must at least be a single bond for each extra residue to 

1570 'bind' them all to a single structure. 

1571 error_code : tup of int 

1572 More information on whether and why the structure was read 

1573 unsuccessfully. See 

1574 :ref:`warning overview<UserGuide_page_warning_overview>` 

1575 for more information - look for code MI_MC_4. 

1576 """ 

1577 

1578 # funcgroup is a list, each item of which corresponds to a different 

1579 # 'option' of what the group could look like. In core.txt, each 

1580 # 'option' is defined on a separate line, in the separate file, each 

1581 # 'option' is marked with the label 'newstruct'. 

1582 all_opts = [] 

1583 all_bonds = [] 

1584 for opt in funcgroup: 

1585 # each opt is a list of strings. Each string is a 'word' in the 

1586 # file. 

1587 ix_count = 0 

1588 residues = [] 

1589 resname = "" 

1590 atoms = [] 

1591 bonds = {} 

1592 for word in opt: 

1593 # if a new residue starts 

1594 if word.startswith("[") and word.endswith("]"): 

1595 # if this is not the first residue 

1596 if resname and atoms: 

1597 atoms = [item.split(",") for item in atoms] 

1598 residues.append([resname.split(","), atoms]) 

1599 atoms = [] 

1600 resname = word[1:-1] 

1601 elif resname or atoms: 

1602 return None, None, (1, len(all_opts), ix_count) 

1603 # if this is the first residue 

1604 else: 

1605 resname = word[1:-1] 

1606 elif word.startswith("[") or word.endswith("]"): 

1607 return None, None, (2, len(all_opts), ix_count) 

1608 # if this is an atom 

1609 else: 

1610 if "(" in word and ")" in word: 

1611 word = word.split("(") 

1612 atnames = word[0] 

1613 word = word[1].split(")") 

1614 numlist = word[0].split(",") 

1615 try: 

1616 bondnums = [int(item) for item in numlist] 

1617 except Exception: 

1618 return None, None, (3, len(all_opts), ix_count) 

1619 for bondnum in bondnums: 

1620 if bondnum in bonds: 

1621 bonds[bondnum].append(ix_count) 

1622 else: 

1623 bonds[bondnum] = [ix_count] 

1624 elif "(" in word or ")" in word: 

1625 return None, None, (4, len(all_opts), ix_count) 

1626 else: 

1627 atnames = word 

1628 

1629 atoms.append(atnames) 

1630 ix_count += 1 

1631 # end of line 

1632 if resname and atoms: 

1633 atoms = [item.split(",") for item in atoms] 

1634 residues.append([resname.split(","), atoms]) 

1635 elif resname or atoms: 

1636 return None, None, (5, len(all_opts), ix_count) 

1637 

1638 bondslist = [] 

1639 

1640 for id, bond in bonds.items(): 

1641 if len(bond) != 2: 

1642 return None, None, (6, len(all_opts), id) 

1643 bondslist.append(bond) 

1644 

1645 # append found info to main lists. 

1646 if residues: 

1647 all_opts.append(residues) 

1648 all_bonds.append(bondslist) 

1649 

1650 return all_opts, all_bonds, (0,)

1743 """Allows the user to select a range of integers to be

1761 for elem in ix_list:

2882 def parse_valid_combinations(self, rawcore, mapdir):
